mod bindings;
use bindings::*;
use spacetimedb_sdk::{DbContext, Table};
use std::time::{Duration, Instant};

// usage: client <token> <seconds> <sql>...
fn main() {
    let args: Vec<String> = std::env::args().collect();
    if args[1] == "bench" {
        return bench(&args[2], args[3].parse().unwrap(), args[4].parse().unwrap(), &args[5]);
    }
    let token = args[1].clone();
    let secs: u64 = args[2].parse().unwrap();
    let queries: Vec<String> = args[3..].to_vec();
    let t0 = Instant::now();
    let conn = DbConnection::builder()
        .with_uri("http://127.0.0.1:3188")
        .with_database_name("viewspike")
        .with_token(Some(token))
        .build()
        .unwrap();
    conn.run_threaded();
    let stamp = move || format!("[{:>6.1}ms]", t0.elapsed().as_secs_f64() * 1000.0);
    macro_rules! hook { ($name:literal, $t:expr, $s:expr) => {{
        let s = $s; $t.on_insert(move |_, r| println!("{} {} INSERT {:?}", s(), $name, (r.id, &r.title)));
        let s = $s; $t.on_delete(move |_, r| println!("{} {} DELETE {:?}", s(), $name, (r.id, &r.title)));
    }}}
    hook!("covered_tasks", conn.db.covered_tasks(), stamp.clone());
    hook!("covered_epics", conn.db.covered_epics(), stamp.clone());
    hook!("visible_epics", conn.db.visible_epics(), stamp.clone());
    hook!("my_created_epics", conn.db.my_created_epics(), stamp.clone());
    let s = stamp.clone();
    conn.subscription_builder()
        .on_applied(move |_| println!("{} APPLIED", s()))
        .on_error(|_, e| println!("ERROR {e}"))
        .subscribe(queries);
    std::thread::sleep(Duration::from_secs(secs));
    let _ = conn.disconnect();
}

// usage: client bench <token> <task_id> <iterations> <query>
fn bench(token: &str, task_id: u64, iters: u32, query: &str) {
    let (tx, rx) = std::sync::mpsc::channel::<(u64, u64)>();
    let (atx, arx) = std::sync::mpsc::channel::<()>();
    let conn = DbConnection::builder()
        .with_uri("http://127.0.0.1:3188")
        .with_database_name("viewspike")
        .with_token(Some(token.to_string()))
        .build()
        .unwrap();
    conn.run_threaded();
    conn.db.covered_tasks().on_insert(move |_, r| { let _ = tx.send((r.id, r.counter)); });
    conn.subscription_builder().on_applied(move |_| { let _ = atx.send(()); }).subscribe([query.to_string()]);
    let t0 = Instant::now();
    arx.recv_timeout(Duration::from_secs(60)).expect("applied");
    let initial = rx.try_iter().count();
    println!("initial subscribe: {:.1} ms, {} rows", t0.elapsed().as_secs_f64() * 1000.0, initial);
    let mut lat = Vec::new();
    let mut last: Option<u64> = None;
    for _ in 0..iters {
        let t = Instant::now();
        conn.reducers.bump_task(task_id).unwrap();
        loop {
            let (id, c) = rx.recv_timeout(Duration::from_secs(30)).expect("delta");
            if id == task_id && last.map_or(true, |l| c == l + 1) { last = Some(c); break; }
        }
        lat.push(t.elapsed().as_secs_f64() * 1000.0);
    }
    lat.sort_by(|a, b| a.partial_cmp(b).unwrap());
    println!("bump->delta ms: min {:.1} p50 {:.1} p95 {:.1} max {:.1}", lat[0], lat[lat.len()/2], lat[lat.len()*95/100], lat[lat.len()-1]);
    let _ = conn.disconnect();
}
